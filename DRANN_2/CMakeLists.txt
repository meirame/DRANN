cmake_minimum_required(VERSION 3.20)
project(DRANN_2 LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(drann_core STATIC DRANN.cpp main.cpp)
set_target_properties(drann_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(drann_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(drann main.cpp)
target_compile_definitions(drann PRIVATE DRANN_CONSOLE)
target_link_libraries(drann PRIVATE drann_core)
include(CTest)
add_executable(drann_tests tests/DRANN_tests.cpp)
target_link_libraries(drann_tests PRIVATE drann_core)
add_test(NAME drann_numerical COMMAND drann_tests)
option(DRANN_BUILD_GODOT "Construir GDExtension" ON)
if(DRANN_BUILD_GODOT)
  include(FetchContent)
  set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
  set(GODOTCPP_BUILD_PROFILE ${CMAKE_CURRENT_SOURCE_DIR}/extension/profile.json CACHE STRING "" FORCE)
  FetchContent_Declare(godot_cpp
    URL https://codeload.github.com/godotengine/godot-cpp/tar.gz/e83fd0904c13356ed1d4c3d09f8bb9132bdc6b77
    URL_HASH SHA256=d4c03daf46c8bef4544614182b0762c3e32789151c7f8f44e8eb2053d5cddeba)
  FetchContent_MakeAvailable(godot_cpp)
  add_library(drann_extension SHARED extension/laboratory.cpp)
  target_link_libraries(drann_extension PRIVATE drann_core godot-cpp)
  set_target_properties(drann_extension PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/godot/bin)
endif()
